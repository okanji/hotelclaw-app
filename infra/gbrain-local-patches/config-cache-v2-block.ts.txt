// HOTELCLAW LOCAL PATCH v2 (2026-07-24; not upstream): TTL'd FULL-TABLE
// memo over config reads. Replaces the upstream getConfig/setConfig/
// unsetConfig methods in src/core/postgres-engine.ts.
//
// Why v2: as of gbrain 0.42.65 the CLI's connectEngine
// (loadConfigWithEngine + reconfigureGatewayWithEngine) issues dozens of
// DISTINCT-key config reads at startup — measured 10.1s for
// loadConfigWithEngine alone from a ~300ms-RTT machine to eu-central-1,
// which trips the CLI's 10s `sources list` connect timeout. The v1 per-key
// memo can't fix a cold start (each distinct key still pays one round
// trip), so v2 prewarms: the first miss inside a TTL window loads the
// WHOLE config table (small, operator-set) in ONE round trip,
// single-flighted; absent keys resolve null from the same snapshot.
// Same-process writes update the cache; cross-process writes are seen
// after the TTL. Disable with GBRAIN_CONFIG_CACHE_TTL_MS=0. Drop when
// upstream ships its batch loader / config cache.

  private _configCache = new Map<string, { v: string | null; at: number }>();
  private _configCacheLoadedAt = 0;
  private _configCacheLoad: Promise<void> | null = null;
  private static _configCacheTtlMs = (() => {
    const n = Number(process.env.GBRAIN_CONFIG_CACHE_TTL_MS);
    return Number.isFinite(n) && n >= 0 ? n : 30_000;
  })();

  private async _prewarmConfigCache(): Promise<void> {
    if (this._configCacheLoad) return this._configCacheLoad;
    this._configCacheLoad = this.connRetry(async () => {
      const rows = await this.sql`SELECT key, value FROM config` as unknown as
        { key: string; value: string }[];
      const at = Date.now();
      this._configCache.clear();
      for (const row of rows) this._configCache.set(row.key, { v: row.value, at });
      this._configCacheLoadedAt = at;
    }).finally(() => {
      this._configCacheLoad = null;
    });
    return this._configCacheLoad;
  }

  async getConfig(key: string): Promise<string | null> {
    const ttl = PostgresEngine._configCacheTtlMs;
    if (ttl > 0) {
      const hit = this._configCache.get(key);
      if (hit && Date.now() - hit.at < ttl) return hit.v;
      // Full snapshot fresh but key absent → genuinely unset: null, no
      // round trip.
      if (Date.now() - this._configCacheLoadedAt < ttl) {
        this._configCache.set(key, { v: null, at: this._configCacheLoadedAt });
        return null;
      }
      // Cold or stale: ONE full-table load (single-flighted), then answer
      // from the snapshot.
      await this._prewarmConfigCache();
      const warm = this._configCache.get(key);
      if (warm) return warm.v;
      this._configCache.set(key, { v: null, at: this._configCacheLoadedAt });
      return null;
    }
    // #1603: a transient pooler drop on this read used to throw / fall
    // through to defaults silently.
    return this.connRetry(async () => {
      const rows = await this.sql`SELECT value FROM config WHERE key = ${key}`;
      return rows.length > 0 ? (rows[0].value as string) : null;
    });
  }

  async setConfig(key: string, value: string): Promise<void> {
    return this.connRetry(async () => {
      await this.sql`
        INSERT INTO config (key, value) VALUES (${key}, ${value})
        ON CONFLICT (key) DO UPDATE SET value = EXCLUDED.value
      `;
      this._configCache.set(key, { v: value, at: Date.now() });
    });
  }

  async unsetConfig(key: string): Promise<number> {
    return this.connRetry(async () => {
      const result = await this.sql`DELETE FROM config WHERE key = ${key}` as unknown as { count: number };
      this._configCache.delete(key);
      return result.count ?? 0;
    });
  }
