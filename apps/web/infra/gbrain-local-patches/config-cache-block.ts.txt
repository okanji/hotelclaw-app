  // HOTELCLAW LOCAL PATCH (2026-07-18, not upstream): short-TTL memo over
  // config reads. mode.ts:loadSearchModeConfig documents "ONE round-trip per
  // knob currently … a future batch loader can collapse this" — a search op
  // issues ~68 getConfig SELECTs, which against a remote (cross-region)
  // Postgres costs ~RTT×68 (~20s measured to eu-central-1). Config keys are
  // operator-set and effectively static per process; a TTL cache restores
  // sub-second search without touching the search path. Same-process writes
  // (setConfig/unsetConfig) update the cache; cross-process writes are seen
  // after the TTL. Disable with GBRAIN_CONFIG_CACHE_TTL_MS=0. Drop this
  // patch when upstream ships the batch loader / config cache.
  private _configCache = new Map<string, { v: string | null; at: number }>();
  private static _configCacheTtlMs = (() => {
    const n = Number(process.env.GBRAIN_CONFIG_CACHE_TTL_MS);
    return Number.isFinite(n) && n >= 0 ? n : 30_000;
  })();

  async getConfig(key: string): Promise<string | null> {
    const ttl = PostgresEngine._configCacheTtlMs;
    if (ttl > 0) {
      const hit = this._configCache.get(key);
      if (hit && Date.now() - hit.at < ttl) return hit.v;
    }
    // #1603: a transient pooler drop on this read used to throw / fall through
    // to defaults silently — which on remote Postgres surfaces as the wrong
    // search mode/knobs and empty-stdout queries.
    return this.connRetry(async () => {
      const rows = await this.sql`SELECT value FROM config WHERE key = ${key}`;
      const v = rows.length > 0 ? (rows[0].value as string) : null;
      if (ttl > 0) this._configCache.set(key, { v, at: Date.now() });
      return v;
    });
  }

  async setConfig(key: string, value: string): Promise<void> {
    return this.connRetry(async () => {
      await this.sql`
        INSERT INTO config (key, value) VALUES (${key}, ${value})
        ON CONFLICT (key) DO UPDATE SET value = EXCLUDED.value
      `;
      this._configCache.set(key, { v: value, at: Date.now() });
    });
  }

  async unsetConfig(key: string): Promise<number> {
    return this.connRetry(async () => {
      const result = await this.sql`DELETE FROM config WHERE key = ${key}` as unknown as { count: number };
      this._configCache.delete(key);
      return result.count ?? 0;
    });
  }
